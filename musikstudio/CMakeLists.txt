cmake_minimum_required(VERSION 3.16)
project(WenuyStudio VERSION 0.4.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE RelWithDebInfo CACHE STRING "Build type" FORCE)
endif()
find_package(Threads REQUIRED)
add_library(wenuystudio_core STATIC src/Wave.cpp src/Engine.cpp src/BlockMixer.cpp src/AudioBuffer.cpp src/ProjectIO.cpp src/History.cpp src/Waveform.cpp src/AudioIO.cpp src/Recovery.cpp src/MediaCheck.cpp src/EditModel.cpp)
target_link_libraries(wenuystudio_core PUBLIC Threads::Threads ${CMAKE_DL_LIBS})
target_include_directories(wenuystudio_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_options(wenuystudio_core PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_executable(wenuy-studio-render src/main.cpp)
target_link_libraries(wenuy-studio-render PRIVATE wenuystudio_core)
add_executable(wenuy-studio-cli src/studio_cli.cpp)
target_link_libraries(wenuy-studio-cli PRIVATE wenuystudio_core)
foreach(t wenuy-studio-render wenuy-studio-cli)
  target_compile_options(${t} PRIVATE -Wall -Wextra -Wpedantic -Werror)
endforeach()
option(WENUY_STUDIO_GUI "Build GTK4 graphical studio when GTK dependencies are present" ON)
option(WENUY_STUDIO_REQUIRE_GUI "Fail configuration when GTK GUI cannot be built" OFF)
option(WENUY_STUDIO_USE_LIBWENUY "Use bundled WenuyOS libwenuy instead of GTK fallback" OFF)
if(WENUY_STUDIO_GUI)
  find_package(PkgConfig QUIET)
  if(PkgConfig_FOUND)
    pkg_check_modules(UI QUIET IMPORTED_TARGET gtk4 libadwaita-1)
  endif()
  if(UI_FOUND)
    add_executable(wenuy-studio gui/StudioGui.cpp)
    target_link_libraries(wenuy-studio PRIVATE wenuystudio_core PkgConfig::UI)
    if(WENUY_STUDIO_USE_LIBWENUY)
      target_compile_definitions(wenuy-studio PRIVATE HAVE_WENUY=1)
      add_subdirectory(third_party/libwenuy)
      target_link_libraries(wenuy-studio PRIVATE wenuy)
    endif()
    target_compile_options(wenuy-studio PRIVATE -Wall -Wextra -Wpedantic -Wno-missing-field-initializers)
    install(TARGETS wenuy-studio RUNTIME DESTINATION bin)
  else()
    if(WENUY_STUDIO_REQUIRE_GUI)
      message(FATAL_ERROR "GTK4/libadwaita development libraries missing; GUI build was required")
    endif()
    message(STATUS "GTK4/libadwaita development libraries missing: building C++ core and CLI only")
  endif()
elseif(WENUY_STUDIO_REQUIRE_GUI)
  message(FATAL_ERROR "GUI was required but WENUY_STUDIO_GUI=OFF")
endif()
