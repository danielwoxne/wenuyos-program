cmake_minimum_required(VERSION 3.16)
project(wenuy-studio VERSION 0.9.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Threads REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(STUDIO_GUI REQUIRED IMPORTED_TARGET gtk4 libadwaita-1)
# Datatyper is optional: regular WAV still works outside a recent WenuyOS SDK.
pkg_check_modules(STUDIO_DT QUIET IMPORTED_TARGET wenuy-datatypes)
add_library(wenuystudio_core STATIC
 src/Wave.cpp src/ProjectIO.cpp src/BlockMixer.cpp src/Playback.cpp src/Engine.cpp
 src/TrackFX.cpp src/Arrangement.cpp src/Sequencer.cpp src/MidiFile.cpp src/EditModel.cpp src/InstrumentBank.cpp src/AudioImport.cpp)
target_include_directories(wenuystudio_core PUBLIC include)
target_link_libraries(wenuystudio_core PUBLIC Threads::Threads ${CMAKE_DL_LIBS})
if(STUDIO_DT_FOUND)
 target_link_libraries(wenuystudio_core PRIVATE PkgConfig::STUDIO_DT)
 target_compile_definitions(wenuystudio_core PRIVATE WENUY_STUDIO_DATATYPES=1)
 message(STATUS "Musikstudio: WenuyOS Datatyper audio enabled")
else()
 message(STATUS "Musikstudio: WAV-only fallback (WenuyOS Datatyper SDK not installed)")
endif()
target_compile_options(wenuystudio_core PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_executable(wenuy-studio gui/GtkStudio.cpp)
target_link_libraries(wenuy-studio PRIVATE wenuystudio_core PkgConfig::STUDIO_GUI)
target_compile_options(wenuy-studio PRIVATE -Wall -Wextra -Wpedantic -Wno-deprecated-declarations)
add_executable(wenuy-studio-cli src/main.cpp)
target_link_libraries(wenuy-studio-cli PRIVATE wenuystudio_core)
target_compile_options(wenuy-studio-cli PRIVATE -Wall -Wextra -Wpedantic -Werror)
install(TARGETS wenuy-studio wenuy-studio-cli RUNTIME DESTINATION bin)
install(DIRECTORY data/ DESTINATION data)
install(FILES wenuyos-app.toml DESTINATION .)
enable_testing()
add_executable(wenuystudio-tests tests/test_core.cpp)
target_link_libraries(wenuystudio-tests PRIVATE wenuystudio_core)
target_compile_options(wenuystudio-tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_test(NAME integrated_070 COMMAND wenuystudio-tests)
add_executable(wenuystudio-050-tests tests/test_sequencer050.cpp)
target_link_libraries(wenuystudio-050-tests PRIVATE wenuystudio_core)
target_compile_options(wenuystudio-050-tests PRIVATE -Wall -Wextra -Wpedantic -Werror -UNDEBUG)
add_test(NAME midi_sequencer_050 COMMAND wenuystudio-050-tests)
add_executable(wenuystudio-060-tests tests/test_060.cpp)
target_link_libraries(wenuystudio-060-tests PRIVATE wenuystudio_core)
target_compile_options(wenuystudio-060-tests PRIVATE -Wall -Wextra -Wpedantic -Werror -UNDEBUG)
add_test(NAME arrangement_fx_060 COMMAND wenuystudio-060-tests)
add_executable(wenuystudio-080-tests tests/test_edit080.cpp)
target_link_libraries(wenuystudio-080-tests PRIVATE wenuystudio_core)
target_compile_options(wenuystudio-080-tests PRIVATE -Wall -Wextra -Wpedantic -Werror -UNDEBUG)
add_test(NAME editing_history_080 COMMAND wenuystudio-080-tests)
add_executable(wenuystudio-090-tests tests/test_piano090.cpp)
target_link_libraries(wenuystudio-090-tests PRIVATE wenuystudio_core)
target_compile_options(wenuystudio-090-tests PRIVATE -Wall -Wextra -Wpedantic -Werror -UNDEBUG)
add_test(NAME piano_pointer_edit_090 COMMAND wenuystudio-090-tests)
add_executable(wenuystudio-guard-tests tests/test_projectguard090.cpp)
target_link_libraries(wenuystudio-guard-tests PRIVATE wenuystudio_core)
target_compile_options(wenuystudio-guard-tests PRIVATE -Wall -Wextra -Wpedantic -Werror -UNDEBUG)
add_test(NAME unsaved_project_guard_090 COMMAND wenuystudio-guard-tests)
add_executable(wenuystudio-instrument-tests tests/test_instruments090.cpp)
target_link_libraries(wenuystudio-instrument-tests PRIVATE wenuystudio_core)
target_compile_options(wenuystudio-instrument-tests PRIVATE -Wall -Wextra -Wpedantic -Werror -UNDEBUG)
add_test(NAME instrument_bank_090 COMMAND wenuystudio-instrument-tests)
add_executable(wenuystudio-audioimport-tests tests/test_audioimport090.cpp)
target_link_libraries(wenuystudio-audioimport-tests PRIVATE wenuystudio_core)
target_compile_options(wenuystudio-audioimport-tests PRIVATE -Wall -Wextra -Wpedantic -Werror -UNDEBUG)
add_test(NAME datatype_audio_import_090 COMMAND wenuystudio-audioimport-tests)
# An optional tool, not a runtime dependency of the GTK music window.
# WenuyGFXConfig.cmake requires json-c + PNG even for the CPU-only target.
# Declare them here so wenuyos verify can install their Debian build packages.
pkg_check_modules(STUDIO_GFX_JSON QUIET IMPORTED_TARGET json-c)
find_package(PNG QUIET)
if(STUDIO_GFX_JSON_FOUND AND PNG_FOUND)
 find_package(WenuyGFX 0.9 CONFIG QUIET)
endif()
if(WenuyGFX_FOUND)
 add_executable(wenuy-studio-gfx tools/GfxNoteMap.cpp)
 target_link_libraries(wenuy-studio-gfx PRIVATE wenuystudio_core WenuyGFX::wenuygfx)
 target_compile_options(wenuy-studio-gfx PRIVATE -Wall -Wextra -Wpedantic -Werror)
 install(TARGETS wenuy-studio-gfx RUNTIME DESTINATION bin)
 add_test(NAME gfx_3d_notes_090
  COMMAND python3 ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_gfxmap090.py
          $<TARGET_FILE:wenuy-studio-gfx>
          ${CMAKE_CURRENT_SOURCE_DIR}/examples/Demo.wstudio)
 message(STATUS "Musikstudio: optional WenuyGFX 0.9 note-map tool enabled")
else()
 message(STATUS "Musikstudio: no WenuyGFX 0.9 SDK; normal music window unchanged")
endif()

add_test(NAME icon_asset_integrity COMMAND python3 ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_icon.py)
