cmake_minimum_required(VERSION 3.20)
project(schack VERSION 0.5.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

add_library(schack_core STATIC src/Chess.cpp src/Lessons.cpp)
target_include_directories(schack_core PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/src")
target_compile_options(schack_core PRIVATE -Wall -Wextra -Wpedantic)

find_package(Threads REQUIRED)
add_library(schack_lan STATIC src/LanChess.cpp)
target_link_libraries(schack_lan PUBLIC schack_core)
target_include_directories(schack_lan PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/src")
target_compile_options(schack_lan PRIVATE -Wall -Wextra -Wpedantic)

enable_testing()
add_executable(schack_tests tests/test_chess.cpp)
target_link_libraries(schack_tests PRIVATE schack_core)
add_test(NAME chess_rules COMMAND schack_tests)
add_executable(schack_lan_tests tests/test_lan.cpp)
target_link_libraries(schack_lan_tests PRIVATE schack_lan Threads::Threads)
add_test(NAME lan_loopback COMMAND schack_lan_tests)
add_test(NAME ui_assets COMMAND python3 ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_ui_assets.py ${CMAKE_CURRENT_SOURCE_DIR})

find_package(PkgConfig REQUIRED)
pkg_check_modules(GTK4 REQUIRED IMPORTED_TARGET gtk4)
pkg_check_modules(WENUY REQUIRED IMPORTED_TARGET wenuy)
add_executable(schack_i18n_tests tests/test_i18n.cpp src/SystemIntegration.cpp)
target_include_directories(schack_i18n_tests PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/src")
target_link_libraries(schack_i18n_tests PRIVATE PkgConfig::WENUY)
target_compile_options(schack_i18n_tests PRIVATE -Wall -Wextra -Wpedantic)
add_test(NAME system_i18n COMMAND schack_i18n_tests)
add_executable(schack src/RuntimeGtk.cpp src/SystemIntegration.cpp)
target_link_libraries(schack PRIVATE schack_lan PkgConfig::GTK4 PkgConfig::WENUY Threads::Threads ${CMAKE_DL_LIBS})
target_compile_options(schack PRIVATE -Wall -Wextra -Wpedantic)
install(TARGETS schack RUNTIME DESTINATION bin)
install(DIRECTORY data/ DESTINATION data)
